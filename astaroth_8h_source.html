69static UNUSED AcCompInfo
70acInitCompInfo()
71{
72 AcCompInfo res;
73 // TP: initially nothing is loaded and if they are not loaded their values
74 // might as well be zero since then a default value is used for them
75 memset(&res, 0, sizeof(res));
76 return res;
77}

79static UNUSED AcMeshInfo
80acInitInfo()
81{
82 AcMeshInfo res;
83 // TP: this is useful for the following reasons:
84 // All enums are initialized by default to the first enum value
85 // All array ptrs are initialized to nulls
86 // All booleans are initialized to false
87 // All booleans about whether values are loaded are false
88 memset(&res, 0, sizeof(res));
89 // memset reads the second parameter as a byte even though it says int in
90 // the function declaration
91 // TP: for backwards compatibility set original datatypes to all ones as before
92 memset(&res.int_params, (uint8_t)0xFF, sizeof(res.int_params));
93 memset(&res.real_params, (uint8_t)0xFF, sizeof(res.real_params));
94 memset(&res.int3_params, (uint8_t)0xFF, sizeof(res.int3_params));
95 memset(&res.real3_params, (uint8_t)0xFF, sizeof(res.real3_params));
96 memset(&res.complex_params, (uint8_t)0xFF, sizeof(res.complex_params));
97
98#if AC_MPI_ENABLED
99 res.comm = (AcCommunicator*)malloc(sizeof(AcCommunicator));
100 res.comm->handle = MPI_COMM_NULL;
101#endif
102 res.run_consts = acInitCompInfo();
103 return res;
104}

123template <typename P, typename V>
124void
125acPushToConfig(AcMeshInfo& config, P param, V val)
126{
127 (void)param;
128 (void)val;
129 if constexpr(IsCompParam<P>())
130 {
131 config.run_consts.config[param] = val;
132 config.run_consts.is_loaded[param] = true;
133 }
134 else if constexpr(IsParam<P>())
135 {
136 config[param] = val;
137 config.is_loaded[param] = true;
138 }
139}

141static AcResult
142acUpdateDecompositionParams(AcMeshInfo* dst)
143{
144#if AC_MPI_ENABLED
145 int nprocs{};
146 int rank{};
147 ERRCHK_ALWAYS(dst->comm != NULL && dst->comm->handle != MPI_COMM_NULL);
148 MPI_Comm_size(dst->comm->handle,&nprocs);
149 MPI_Comm_rank(dst->comm->handle,&rank);
150 const int3 decomp = acDecompose(nprocs,*dst);
151 const int3 pid3d = acGetPid3D(rank,decomp,*dst);
152 acPushToConfig((*dst),AC_domain_coordinates,pid3d);
153 acPushToConfig((*dst),AC_domain_decomposition,decomp);
154 return AC_SUCCESS;
155#else
156 return AC_FAILURE;
157#endif
158}
