373static inline AcMeshDims
374acGetMeshDims(const AcMeshInfo info, const VertexBufferHandle vtxbuf)
375{
376 const int3 halos = acGetFieldHalos(info,vtxbuf);
377 const Volume n0 =
378 (Volume)
379 {
380 as_size_t(halos.x),
381 as_size_t(halos.y),
382 as_size_t(halos.z)
383 };
384 const Volume m1 =
385 (Volume){
386 as_size_t(info.int3_params[vtxbuf_dims[vtxbuf]].x),
387 as_size_t(info.int3_params[vtxbuf_dims[vtxbuf]].y),
388 as_size_t(info.int3_params[vtxbuf_dims[vtxbuf]].z)
389 };
390 const Volume n1 =
391 (Volume)
392 {
393 m1.x-n0.x,
394 m1.y-n0.y,
395 m1.z-n0.z,
396 };
397 const Volume m0 = (Volume){0, 0, 0};
398 const Volume nn =
399 (Volume)
400 {
401 m1.x-2*n0.x,
402 m1.y-2*n0.y,
403 m1.z-2*n0.z,
404 };
405 const Volume reduction_tile = (Volume)
406 {
407 as_size_t(info.int3_params[AC_reduction_tile_dimensions].x),
408 as_size_t(info.int3_params[AC_reduction_tile_dimensions].y),
409 as_size_t(info.int3_params[AC_reduction_tile_dimensions].z)
410 };
411
412 return (AcMeshDims){
413 .n0 = n0,
414 .n1 = n1,
415 .m0 = m0,
416 .m1 = m1,
417 .nn = nn,
418 .reduction_tile = reduction_tile,
419 };
420}

446template <typename P, typename V>
447void
448acPushToConfig(AcMeshInfo& config, P param, V val)
449{
450 (void)param;
451 (void)val;
452 if constexpr(IsCompParam<P>())
453 {
454 config.run_consts.config[param] = val;
455 config.run_consts.is_loaded[param] = true;
456 }
457 else if constexpr(IsParam<P>())
458 {
459 config[param] = val;
460 config.is_loaded[param] = true;
461 }
462}

466{
467#if AC_MPI_ENABLED
468 int nprocs{};

475 acPushToConfig((*dst),AC_domain_coordinates,pid3d);
476 acPushToConfig((*dst),AC_domain_decomposition,decomp);
477 return AC_SUCCESS;
478#else
479 return AC_FAILURE;
480#endif
481}

485 static UNUSED AcCompInfo acInitCompInfo()
486 {
487 AcCompInfo res;
488 //TP: initially nothing is loaded and if they are not loaded their values
489 //might as well be zero since then a default value is used for them
490 memset(&res,0,sizeof(res));
491 return res;
492 }
493 static UNUSED AcMeshInfo acInitInfo()
494 {
495 AcMeshInfo res;
496 //TP: this is useful for the following reasons:
497 //All enums are initialized by default to the first enum value
498 //All array ptrs are initialized to nulls
499 //All booleans are initialized to false
500 //All booleans about whether values are loaded are false
501 memset(&res,0,sizeof(res));
502 // memset reads the second parameter as a byte even though it says int in
503 // the function declaration
504 //TP: for backwards compatibility set original datatypes to all ones as before
505 memset(&res.int_params, (uint8_t)0xFF, sizeof(res.int_params));
506 memset(&res.real_params, (uint8_t)0xFF, sizeof(res.real_params));
507 memset(&res.int3_params, (uint8_t)0xFF, sizeof(res.int3_params));
508 memset(&res.real3_params, (uint8_t)0xFF, sizeof(res.real3_params));
509 memset(&res.complex_params, (uint8_t)0xFF, sizeof(res.complex_params));
510
511#if AC_MPI_ENABLED
512 res.comm = (AcCommunicator*)malloc(sizeof(AcCommunicator));
513 res.comm->handle = MPI_COMM_NULL;
514#endif
515 res.run_consts = acInitCompInfo();
516 return res;
517 }
518 static UNUSED AcMesh acInitMesh()

524 }
525 res.info = acInitInfo();
526 return res;
527 }
