2//TP: handling of constant loops at the start of generate (unroll_constant_loops), included in codegen.c after the AST helpers it uses
3// (is_for_iteration_statement, replace_variable_with_constant_int, create_func_call, push_to_statement_list, get_nodes, eval_int, ...).
4// Short loops are unrolled as before, long loops are kept rolled with their iteration-specific stencil calls hoisted (see below).

6#ifndef MAX_UNROLL_ITERATIONS
7#define MAX_UNROLL_ITERATIONS (4)
8#endif
9//TP: long constant loops (more than MAX_UNROLL_ITERATIONS iterations), e.g. loops over 32 dust species.
10// Unrolling them makes every iteration-specific stencil call a call on a compile-time known field, so all of them are preloaded
11// at the start of the kernel and the kernels spill heavily to scratch.
12// Instead the loop is kept rolled and the stencil calls that depend only on the loop variable are hoisted to locals at the start of each
13// iteration between ac_dynamic_stencils_begin() and ac_dynamic_stencils_end(). The analysis marks the stencil calls in between
14// with AC_STENCIL_DYNAMIC_CALL and stencilgen computes the stencils that are called only like this on demand instead of preloading them.
15// So the loads of one iteration are still issued together while the working set is that of one iteration.
16// If the arrays indexed in the hoisted calls are used also outside long loops the stencils are preloaded anyways so the loop is unrolled as before.
17typedef struct
18{
19 string_vec stencils;
20 string_vec funcs;
21 node_vec func_callees; //TP: data[i] is actually a string_vec* of the callees of funcs[i]
22 node_vec func_stencils; //TP: data[i] is actually a string_vec* of the stencils funcs[i] calls directly or through other functions
23 bool* has_side_effects;
24 struct hashmap_s used_outside_long_loops; //TP: keys "array:stencil" of the stencils called on the elements of the array outside long loops
25 int hoist_counter;
26} loop_hoist_info;

63//TP: -1 if not a for loop over a range with bounds known at compile time.
64// *literal_bounds tells whether the bounds are int literals: only those loops are unrolled, the others are rolled as before
65// but they can be long loops of which the iteration-specific calls are hoisted, e.g. for k in 1:ndustspec+1
66static int
67get_const_loop_trip_count_base(const ASTNode* node, bool* literal_bounds, int* start_value)
68{
69 if(!is_for_iteration_statement(node)) return -1;
70 const ASTNode* for_expr = node->lhs->rhs;
71 if(!for_expr || !for_expr->rhs || for_expr->rhs->token != RANGE) return -1;
72 const ASTNode* start_node = for_expr->rhs->lhs->lhs;
73 const ASTNode* end_node = for_expr->rhs->rhs->lhs;
74 if(!start_node || !end_node) return -1;
75 *literal_bounds = is_number(intern(combine_buffers_new(start_node))) && is_number(intern(combine_buffers_new(end_node)));
76 int start = 0, end = 0;
77 if(!eval_loop_bound(start_node,&start) || !eval_loop_bound(end_node,&end)) return -1;
78 *start_value = start;
79 return end > start ? end-start : 0;
80}

120//TP: an identifier in the arguments of a call is marked as used with the stencils of the callee,
121// an identifier outside calls, e.g. F_DUST_DENSITY in nd = F_DUST_DENSITY, with the value stencil.
122// Conservative: all identifiers are marked, not only fields
123static void
124collect_stencil_uses_outside_long_loops(const ASTNode* node, const bool in_function, const bool in_call_args, loop_hoist_info* info)
125{
126 if(is_long_const_loop(node)) return;
127 const bool in_func = in_function || (node->type & NODE_FUNCTION);
128 if(in_func && !in_call_args && node->token == IDENTIFIER && node->buffer)
129 mark_used_outside_long_loops(info,node->buffer,intern("value_stencil"));
130 const char* name = in_func ? get_func_call_name(node) : NULL;
131 if(name && node->rhs)
132 {
133 string_vec tmp = VEC_INITIALIZER;
134 const string_vec* stencils = get_callee_stencils(info,name,&tmp);
135 if(stencils->size)
136 {
137 string_vec ids = VEC_INITIALIZER;
138 collect_identifiers(node->rhs,&ids);
139 for(size_t i = 0; i < ids.size; ++i)
140 for(size_t j = 0; j < stencils->size; ++j)
141 mark_used_outside_long_loops(info,ids.data[i],stencils->data[j]);
142 free_str_vec(&ids);
143 }
144 free_str_vec(&tmp);
145 }
146 if(node->lhs) collect_stencil_uses_outside_long_loops(node->lhs,in_func,in_call_args,info);
147 if(node->rhs) collect_stencil_uses_outside_long_loops(node->rhs,in_func,in_call_args || name != NULL,info);
148}
149static loop_hoist_info
150get_loop_hoist_info(const ASTNode* root)
151{
152 loop_hoist_info info;
153 memset(&info,0,sizeof(info));
154 {
155 node_vec nodes = VEC_INITIALIZER;
156 get_nodes(root,&nodes,&info.stencils,NODE_STENCIL);
157 free_node_vec(&nodes);
158 }
159 node_vec funcs = VEC_INITIALIZER;
160 string_vec func_names = VEC_INITIALIZER;
161 get_nodes(root,&funcs,&func_names,NODE_DFUNCTION);
162 //TP: overloads share the name so the info of a name is the union over them
163 for(size_t i = 0; i < funcs.size; ++i)
164 {
165 int index = str_vec_get_index(info.funcs,func_names.data[i]);
166 if(index < 0)
167 {
168 push(&info.funcs,func_names.data[i]);
169 string_vec* callees = calloc(1,sizeof(string_vec));
170 push_node(&info.func_callees,(const ASTNode*)callees);
171 index = info.funcs.size-1;
172 }
173 collect_callee_names(funcs.data[i]->rhs ? funcs.data[i]->rhs : funcs.data[i],(string_vec*)info.func_callees.data[index]);
174 }
175 free_node_vec(&funcs);
176 free_str_vec(&func_names);
177 const size_t n = info.funcs.size;
178 for(size_t i = 0; i < n; ++i) push_node(&info.func_stencils,(const ASTNode*)calloc(1,sizeof(string_vec)));
179 info.has_side_effects = calloc(n+1,sizeof(bool));
180 bool changed = true;
181 while(changed)
182 {
183 changed = false;
184 for(size_t i = 0; i < n; ++i)
185 {
186 const string_vec* callees = (const string_vec*)info.func_callees.data[i];
187 for(size_t j = 0; j < callees->size; ++j)
188 {
189 const char* callee = callees->data[j];
190 const int index = str_vec_get_index(info.funcs,callee);
191 string_vec* own_stencils = (string_vec*)info.func_stencils.data[i];
192 if(str_vec_get_index(info.stencils,callee) >= 0 && str_vec_get_index(*own_stencils,callee) < 0) { push(own_stencils,callee); changed = true; }
193 if(index >= 0 && index != (int)i)
194 {
195 const string_vec* callee_stencils = (const string_vec*)info.func_stencils.data[index];
196 for(size_t k = 0; k < callee_stencils->size; ++k)
197 if(str_vec_get_index(*own_stencils,callee_stencils->data[k]) < 0) { push(own_stencils,callee_stencils->data[k]); changed = true; }
198 }
199 const bool side_effects = is_side_effect_call_name(callee) || (index >= 0 && info.has_side_effects[index]);
200 if(side_effects && !info.has_side_effects[i]) { info.has_side_effects[i] = true; changed = true; }
201 }
202 }
203 }
204 hashmap_create(4096,&info.used_outside_long_loops);
205 collect_stencil_uses_outside_long_loops(root,false,false,&info);
206 return info;
207}

310//TP: goes through the loop body in program order and stops hoisting at the first write since the calls could read the written values.
311// Calls in conditionally executed code are not hoisted since they could be invalid if executed unconditionally,
312// except under uniform conditions (see is_uniform_condition): if such a condition is false the hoisted call is only unneeded work
313// that the compiler removes since the result is not used. The field the call is on could be invalid if the condition is false
314// but on the device the stencil functions return NAN for an unknown field and the analysis skips invalid fields.
315static void
316find_hoistable_calls(const ASTNode* node, const char* loop_variable, const string_vec assigned, const loop_hoist_info* info, const bool conditional, bool* write_seen, node_vec* dst)
317{
318 if(*write_seen) return;
319 if((node->type & NODE_FUNCTION_CALL) && !conditional && is_hoistable_call(node,loop_variable,assigned,info))
320 {
321 push_node(dst,node);
322 return;
323 }
324 if(node->type & NODE_IF)
325 {
326 const bool uniform = node->lhs && is_uniform_condition(node->lhs,loop_variable,assigned);
327 if(node->lhs) find_hoistable_calls(node->lhs,loop_variable,assigned,info,conditional,write_seen,dst);
328 if(node->rhs) find_hoistable_calls(node->rhs,loop_variable,assigned,info,conditional || !uniform,write_seen,dst);
329 return;
330 }
331 const bool cond = conditional || (node->type & NODE_TERNARY) || is_loop_statement(node);
332 if(node->lhs) find_hoistable_calls(node->lhs,loop_variable,assigned,info,cond,write_seen,dst);
333 if(node->rhs) find_hoistable_calls(node->rhs,loop_variable,assigned,info,cond || is_logical_binary_expression(node),write_seen,dst);
334 const char* name = get_func_call_name(node);
335 if(name)
336 {
337 const int index = str_vec_get_index(info->funcs,name);
338 if(is_side_effect_call_name(name) || (index >= 0 && info->has_side_effects[index])) *write_seen = true;
339 }
340}

351typedef enum { LOOP_UNROLL, LOOP_KEEP_ROLLED, LOOP_HOISTED } long_loop_action;
352static long_loop_action
353hoist_iteration_specific_calls(ASTNode* node, loop_hoist_info* info)
354{
355 ASTNode* body = node->rhs;
356 if(!body || !(body->type & NODE_BEGIN_SCOPE) || !body->lhs || !(body->lhs->type & NODE_STATEMENT_LIST_HEAD)) return LOOP_KEEP_ROLLED;
357 const char* loop_variable = node->lhs->rhs->lhs->buffer;
358 string_vec assigned = VEC_INITIALIZER;
359 collect_assigned_identifiers(body,&assigned);
360 node_vec calls = VEC_INITIALIZER;
361 bool write_seen = false;
362 find_hoistable_calls(body,loop_variable,assigned,info,false,&write_seen,&calls);
363 free_str_vec(&assigned);
364 if(calls.size == 0)
365 {
366 free_node_vec(&calls);
367 return LOOP_KEEP_ROLLED;
368 }
369 //TP: unrolled as before if all the stencils of the hoisted calls are called on the same arrays also outside long loops
370 bool used_elsewhere = true;
371 for(size_t i = 0; i < calls.size && used_elsewhere; ++i)
372 {
373 string_vec arrays = VEC_INITIALIZER;
374 collect_loop_indexed_arrays(calls.data[i]->rhs,loop_variable,&arrays);
375 string_vec tmp = VEC_INITIALIZER;
376 const string_vec* stencils = get_callee_stencils(info,get_func_call_name(calls.data[i]),&tmp);
377 if(arrays.size == 0) used_elsewhere = false;
378 for(size_t a = 0; a < arrays.size && used_elsewhere; ++a)
379 for(size_t st = 0; st < stencils->size && used_elsewhere; ++st)
380 if(!used_outside_long_loops(info,arrays.data[a],stencils->data[st])) used_elsewhere = false;
381 free_str_vec(&tmp);
382 free_str_vec(&arrays);
383 }
384 if(used_elsewhere)
385 {
386 free_node_vec(&calls);
387 return LOOP_UNROLL;
388 }
389 ASTNode* head = body->lhs;
390 while(head->lhs && (head->lhs->type & NODE_STATEMENT_LIST_HEAD)) head = head->lhs;
391 push_to_statement_list(head,create_non_returning_call_statement("ac_dynamic_stencils_begin"));
392 //TP: the same call is hoisted only once
393 string_vec hoisted_calls = VEC_INITIALIZER;
394 string_vec hoisted_vars = VEC_INITIALIZER;
395 for(size_t i = 0; i < calls.size; ++i)
396 {
397 ASTNode* call = (ASTNode*)calls.data[i];
398 const char* call_str = intern(combine_all_new(call));
399 const int index = str_vec_get_index(hoisted_calls,call_str);
400 if(index >= 0)
401 {
402 replace_node(call,create_primary_expression(hoisted_vars.data[index]));
403 continue;
404 }
405 //TP: numbered per function such that the otherwise identical copies of a kernel stay identical (see dedup_optimized_kernels)
406 const char* var_name = sprintf_intern("ac_loop_hoisted_%d",info->hoist_counter++);
407 //TP: typed such that remove_dead_writes can remove the hoisted calls that are not needed
408 const char* type = get_expr_type(call);
409 ASTNode* call_copy = astnode_dup(call,NULL);
410 replace_node(call,create_primary_expression(var_name));
411 push_to_statement_list(head,create_assignment(create_declaration(var_name,type,CONST_STR),call_copy,EQ_STR));
412 push(&hoisted_calls,call_str);
413 push(&hoisted_vars,var_name);
414 }
415 free_str_vec(&hoisted_calls);
416 free_str_vec(&hoisted_vars);
417 push_to_statement_list(head,create_non_returning_call_statement("ac_dynamic_stencils_end"));
418 free_node_vec(&calls);
419 return LOOP_HOISTED;
420}
421static void
422unroll_constant_loops_recursive(ASTNode* node, loop_hoist_info* info)
423{
424 if(node->type & NODE_FUNCTION)
425 {
426 const int counter = info->hoist_counter;
427 info->hoist_counter = 0;
428 if(node->lhs) unroll_constant_loops_recursive(node->lhs,info);
429 if(node->rhs) unroll_constant_loops_recursive(node->rhs,info);
430 info->hoist_counter = counter;
431 return;
432 }
433 TRAVERSE_PREAMBLE_PARAMS(unroll_constant_loops_recursive,info);
434 bool literal_bounds = false;
435 int start_value = 0;
436 const int trip_count = get_const_loop_trip_count_base(node,&literal_bounds,&start_value);
437 if(trip_count < 0) return;
438 if(trip_count > MAX_UNROLL_ITERATIONS && hoist_iteration_specific_calls(node,info) != LOOP_UNROLL) return;
439 if(!literal_bounds) return;
440 const ASTNode* for_expr = node->lhs->rhs;
441 const char* loop_variable = for_expr->lhs->buffer;
442 const ASTNode* iterated_body = node->rhs;
443 //TP: a compound body has to stay a scope also for the DSL: with only the {} emitted for the C++ compiler
444 //the declarations of the later iterations are generated as assignments to variables
445 //that the C++ compiler sees declared only inside the {} of the first iteration
446 //if(iterated_body->type & NODE_BEGIN_SCOPE && iterated_body->lhs) iterated_body = iterated_body->lhs;
447
448 const int end_value = start_value + trip_count;
449 node_vec nodes = VEC_INITIALIZER;
450 for(int i = start_value; i < end_value; ++i)
451 {
452 const char* replacement = intern(itoa(i));
453 ASTNode* res = astnode_dup(iterated_body,NULL);
