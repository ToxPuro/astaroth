396static UNUSED size_t
397prof_count(const Profile prof, const size3_t counts)
398{
399 if(NUM_PROFILES == 0) return 0;
400 return
401 prof_types[prof] == PROFILE_X ? counts.x :
402 prof_types[prof] == PROFILE_Y ? counts.y :
403 prof_types[prof] == PROFILE_Z ? counts.z :
404 prof_types[prof] == PROFILE_XY ? counts.x*counts.y :
405 prof_types[prof] == PROFILE_XZ ? counts.x*counts.z :
406 prof_types[prof] == PROFILE_YX ? counts.y*counts.x :
407 prof_types[prof] == PROFILE_YZ ? counts.y*counts.z :
408 prof_types[prof] == PROFILE_ZX ? counts.z*counts.x :
409 prof_types[prof] == PROFILE_ZY ? counts.z*counts.y :
410 0;
411}

637 template <typename P>
638 constexpr static auto
639 get_array_dim_sizes(const P array, const AcMeshInfo info)
640 {
641 auto dims = get_array_info(array).dims;
642 int num_dims = get_array_info(array).num_dims;
643 std::array<size_t,20> res{};
644 for(int i = 0; i < num_dims; ++i)
645 {
646 if(!dims[i].from_config)
647 {
648 res[i] = dims[i].base;
649 continue;
650 }
651 if(dims[i].member == NULL)
652 {
653 if(dims[i].run_const)
654 {
655 res[i] = info.run_consts.config.int_params[dims[i].base];
656 }
657 else
658 {
659 res[i] = info.int_params[dims[i].base];
660 }
661 continue;
662 }
663 if(!strcmp(dims[i].member,"x"))
664 {
665 res[i] = info.int3_params[dims[i].base].x;
666 continue;
667 }
668 if(!strcmp(dims[i].member,"y"))
669 {
670 res[i] = info.int3_params[dims[i].base].y;
671 continue;
672 }
673 if(!strcmp(dims[i].member,"z"))
674 {
675 res[i] = info.int3_params[dims[i].base].z;
676 continue;
677 }
678 }
679 return res;
680 }
